```bash
python main.py \
    --bs 80 \                          # 批次大小

    --out_dir checkpoints/$EXP_NAME \  # 输出目录
    --use_amp                          # 混合精度训练
```

### 3.2 模型代码定义

```python
# ClipCap.py

class ClipCaptionModel(nn.Module):

    def __init__(
        self,
        continuous_length: int = 10,       # Soft Prompt长度
        clip_project_length: int = 10,     # CLIP特征投影长度
        clip_hidden_size: int = 512,       # CLIP特征维度 (ViT-B/32)
        num_layers: int = 8,               # MappingNetwork层数
        num_heads: int = 8,                # 注意力头数
        gpt_type: str = 'gpt2',            # 语言模型类型
        soft_prompt_first: bool = False,   # Soft Prompt是否在前
        only_hard_prompt: bool = False     # 是否仅使用Hard Prompt
    ) -> None:
        super(ClipCaptionModel, self).__init__()
        self.soft_prompt_first = soft_prompt_first
        self.only_hard_prompt = only_hard_prompt
        self.continuous_length = continuous_length
        
        # 加载GPT-2语言模型
        self.gpt, self.gpt_hidden_size = get_language_mode(gpt_type)
        
        # 创建映射网络
        self.mapping_network = MappingNetwork(
            clip_project_length, 
            clip_hidden_size, 
            continuous_length, 
            self.gpt_hidden_size, 
            num_layers, 
            num_heads
        )
```

### 3.3 MappingNetwork 详解

```python
# ClipCap.py

class MappingNetwork(nn.Module):

    def __init__(
        self,
        clip_project_length: int,    # CLIP特征投影长度
        clip_hidden_size: int,       # CLIP特征维度
        prefix_length: int,          # 输出的Soft Prompt长度
        d_model: int,                # GPT-2隐藏层维度
        num_layers: int = 8,
        num_heads: int = 8
    ) -> None:
        super(MappingNetwork, self).__init__()
        self.clip_project_length = clip_project_length
        
        # 线性投影层: CLIP特征 → GPT-2 embedding空间
        self.linear = nn.Linear(clip_hidden_size, clip_project_length * d_model)
        
        # 可学习的前缀嵌入
        self.prefix_const = nn.Parameter(
            torch.randn(prefix_length, d_model), 
            requires_grad=True
        )
        
        # Transformer用于特征融合
        self.transformer = Transformer(d_model, num_layers, num_heads)
        
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: CLIP特征 (batch_size, clip_hidden_size)
        Return:
            Soft Prompt嵌入 (batch_size, prefix_length, d_model)
        """
        # 投影CLIP特征
        x = self.linear(x).view(x.shape[0], self.clip_project_length, -1)
        
        # 扩展可学习前缀
        prefix = self.prefix_const.unsqueeze(0).expand(x.shape[0], *self.prefix_const.shape)
        
        # 拼接并通过Transformer
        inputs = torch.cat((x, prefix), dim=1)
        outputs = self.transformer(inputs)[:, self.clip_project_length:, :]
        
        return outputs
```

### 4.1 Text-Only Training（仅文本训练）

ViECap的核心创新之一是**仅使用文本数据进行训练**，不需要图像-文本配对数据：

```

├─────────────────────────────────────────────────────────────────────────────────┤

│         ↓                                                                        │
│   ┌─────────────────────────────────────────────────────────────────────────┐   │
│   │                      CLIP 文本编码器                                     │   │
│   │   text_features = CLIP.encode_text(captions)  # (batch, 512)           │   │
│   └─────────────────────────────────────────────────────────────────────────┘   │
│         ↓                                                                        │
│   ┌─────────────────────────────────────────────────────────────────────────┐   │
│   │                        噪声注入                                          │   │

│   │   • 模拟图像特征与文本特征的分布差异                                       │   │
```

### 4.2 噪声注入机制

```python
# utils.py

def noise_injection(x, variance=0.001, device='cuda:0') -> torch.Tensor:
    """
    向特征添加高斯噪声，模拟图像特征与文本特征的分布差异
    
    Args:
        x: 输入特征 (batch_size, clip_hidden_size)
        variance: 噪声方差
    Return:
        添加噪声后的特征
    """
    if variance == 0.0:
        return x
    std = math.sqrt(variance)
    
    # 先归一化
    x = torch.nn.functional.normalize(x, dim=-1)
    
    # 添加高斯噪声
    x = x + (torch.randn(x.shape, device=device) * std)
    
    # 再次归一化
    return torch.nn.functional.normalize(x, dim=-1)
```

**为什么需要噪声注入？**

在CLIP的特征空间中，图像特征和文本特征虽然对齐，但仍存在一定的分布差异（modality gap）。通过添加噪声：
- 训练时：文本特征 + 噪声 ≈ 模拟图像特征
- 推理时：真实图像特征可以被模型正确处理

### 4.3 实体随机遮蔽（Entity Random Masking）

```python
# utils.py

def entities_process(
    args,
    detected_entities: List[str],  # [man, dog, park]
    stopwords: List[str],
    people_vocabs: List[str],
    objects_vocabs: List[str],
) -> List[str]:
    """
    处理检测到的实体，支持随机遮蔽
    """
    process_entities = []
    for i in range(len(detected_entities)):
        if i >= args.max_num_of_entities:
            break
        detected_entity = detected_entities[i]
        
        # 人物类别统一为 'person'
        if detected_entity in people_vocabs:
            detected_entity = 'person'
        elif len(detected_entity) > 1 and detected_entity not in stopwords:
            pass
        else:
            continue

        if args.random_mask:
            random_prob = random.random()
            if random_prob < args.prob_of_random_mask:
                pass  # 遮蔽该实体（不添加到列表）
            else:
                process_entities.append(detected_entity)
        else:
            return detected_entities
    
    return process_entities
```

**随机遮蔽的作用**：
- 防止模型过度依赖Hard Prompt中的实体
- 增强模型的泛化能力
- 默认遮蔽概率：40%

## 五、训练循环代码解析

### 5.1 主训练循环

```python
# main.py

def train(args, datasets, model, warmup_steps=5000, output_dir='.', output_prefix=''):
    device = args.device
    batch_size = args.bs
    epochs = args.epochs
    
    # 创建输出目录
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    model = model.to(device)
    model.train()
    
    # 优化器和调度器
    optimizer = AdamW(model.parameters(), lr=args.lr)
    dataloader = DataLoader(datasets, batch_size=batch_size, shuffle=True, 
                           drop_last=True, collate_fn=collate)
    scheduler = get_linear_schedule_with_warmup(
        optimizer, 
        num_warmup_steps=warmup_steps, 
        num_training_steps=epochs * len(dataloader)
    )
    scaler = torch.cuda.amp.GradScaler(enabled=args.use_amp)
    
    for epoch in range(epochs):
        print(f">>> Training epoch {epoch}")
        progress = tqdm(total=len(dataloader), desc=output_prefix)
        
        for idx, (captions_clip, captions_gpt_tokens, captions_tokens_for_loss, 
                  masks, hard_prompts_length) in enumerate(dataloader):
            
            model.zero_grad()
            
            # 获取CLIP特征并添加噪声
            continuous_prefix = captions_clip.to(device).float()
            if args.normalize_prefix:
                continuous_prefix /= continuous_prefix.norm(2, dim=-1, keepdim=True)
            continuous_prefix = noise_injection(continuous_prefix, 
                                               variance=args.noise_variance, 
                                               device=args.device)
            
            # 前向传播
            with torch.cuda.amp.autocast(enabled=args.use_amp):
                if args.using_hard_prompt:
                    outputs = model(continuous_prefix, captions_gpt_tokens, 
                                   hard_prompts_length, masks)
                else:
                    outputs = model(continuous_prefix, captions_gpt_tokens, mask=masks)
                logits = outputs.logits
            
            # 计算损失
            captions_tokens_for_loss = captions_tokens_for_loss.masked_fill(
                captions_tokens_for_loss == tokenizer.eos_token_id, 0
            )
            loss = nnf.cross_entropy(
                logits.reshape(-1, logits.shape[-1]), 
                captions_tokens_for_loss.flatten(), 
                ignore_index=0
            )
            
            # 反向传播
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            scheduler.step()
            optimizer.zero_grad()
            
            progress.update()
        
        progress.close()
        
        # 保存权重
        if (epoch+1) % args.save_every == 0 or epoch == epochs - 1:
            ckpt_path = os.path.join(output_dir, f"{output_prefix}-00{epoch}.pt")
            torch.save(model.state_dict(), ckpt_path)
            print(f'saving checkpoint to {ckpt_path}')
```
